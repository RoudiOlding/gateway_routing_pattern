# Construir imágenes de Docker
docker build -t product-service ./services/product-service
docker build -t user-service ./services/user-service
docker build -t order-service ./services/order-service
docker build -t notification-service ./services/notification-service

# Desplegar Kubernetes
kubectl apply -f k8s/

# Esperar por los deployments
kubectl rollout status deployment/product-service-v1
kubectl rollout status deployment/product-service-v2
kubectl rollout status deployment/user-service
kubectl rollout status deployment/order-service
kubectl rollout status deployment/notification-service

# Prueba simple de un Endpoint Routing
curl http://localhost/api/products
curl http://localhost/api/users
curl http://localhost/api/orders
curl http://localhost/api/notifications

# Prueba de versiones de rutas (Product Service)
curl http://localhost/api/products -H "x-api-version: v1"
curl http://localhost/api/products -H "x-api-version: v2"
curl http://localhost/api/v2/products

# Prueba de Load Balancing (Multiples consultas)
for i in {1..5}; do
  echo "Request $i:"
  curl http://localhost/api/users | grep instance
  sleep 1
done

# Prueba de Alta Disponibilidad - Eliminar un user service pod
kubectl get pods -l app=user-service
kubectl delete pod [POD_NAME]

# Seguir probando mientras el pod fue eliminado
for i in {1..3}; do
  echo "Request during failover $i:"
  curl http://localhost/api/users | grep instance
  sleep 2
done

# Visualizar el estado de todos los pods
kubectl get pods -o wide

# Visualizar services
kubectl get svc

# Visualizar ingress
kubectl get ingress

# Verificar log de los pods
kubectl logs -l app=product-service
kubectl logs -l app=user-service
kubectl logs -l app=order-service
kubectl logs -l app=notification-service

# Escalar servicios
kubectl scale deployment user-service --replicas=5
kubectl scale deployment order-service --replicas=3

# Limpieza
kubectl delete -f k8s/
